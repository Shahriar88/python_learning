# Git LFS: Upload large files
#==============================================================

# Install Git LFS
conda install -c conda-forge git-lfs -y

# Initialize Git LFS
git lfs install

# Go to repository
cd /d C:\Users\kec994\Documents\LDED

# Track a new ZIP file using Git LFS
git lfs track "Primary_Setup_Oct16_2026.zip"

# Stage LFS configuration and ZIP
git add .gitattributes "Primary_Setup_Oct16_2026.zip"

# Check status
git status

# Commit
git commit -m "Track setup ZIP with Git LFS"

# Upload to GitHub
git push origin main

# Verify LFS tracking
git lfs ls-files


#==============================================================
# Git LFS: Migrate a previously committed large file
#==============================================================
# Only use if the file was previously committed as regular Git.
# WARNING: Migration can rewrite Git history.
# Back up your repository before running migration.

# Inspect the file history
git log --oneline --all -- "Primary_Setup_Oct16_2026.zip"

# Inspect migration information
git lfs migrate info --everything

# Migrate the ZIP into Git LFS (history rewriting)
git lfs migrate import --include="Primary_Setup_Oct16_2026.zip" --everything

# Verify changes
git status
git log -5 --oneline
git lfs ls-files

# A published repository may require a coordinated
# force-with-lease push after migration.
# Do not force-push without checking collaborators
# and preserving backups.

#==============================================================
# IMPORTANT NOTES
#==============================================================
# 1. git clone is generally needed only once.
# 2. git add . stages new, modified, and deleted files.
# 3. git commit saves staged changes locally.
# 4. git push sends committed changes to GitHub.
# 5. git pull retrieves and integrates remote changes.
# 6. git restore --staged . unstages without discarding edits.
# 7. git restore . discards uncommitted tracked-file edits.
# 8. git lfs track applies to new staged file versions.
# 9. Git LFS migration may rewrite previous commits.
# 10. Never upload private credentials or API keys.
#==============================================================

# EXAMPLE USAGE 1: Upload a NEW large ZIP file with Git LFS

# Assumptions: Repository already cloned; file copied into LDED;

# Enable LFS locally (once per computer) and track this exact ZIP

# Stage LFS rules and the new ZIP file

# Commit locally, then upload to GitHub

# Note: If this ZIP was previously committed WITHOUT LFS,
# tracking it now does not remove historical ordinary Git blobs.
# Do NOT run 'git lfs migrate import --everything' casually.
